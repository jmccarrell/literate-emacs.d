* General Settings
** Native compilation on macOS

   Temporary; remove once [[https://github.com/caldwell/build-emacs/issues/158][build-emacs#158]] ships a fixed Emacs.app.  The GCC
   bundled in the emacsformacosx.com build derives =-mmacosx-version-min=
   from the kernel version minus 9, which on macOS 27 gives =18.0=; Apple's
   clang rejects it, so every native compile fails.  An explicit version for
   the native compiler's driver overrides it without touching the
   environment of other processes; =11= is the minimum this build targets.
   Background compiles hand the option to their workers only if =comp= is
   loaded, and they load just =comp-run=, so =comp-run= pulls =comp= in.

  #+BEGIN_SRC emacs-lisp
    (when (eq system-type 'darwin)
      (with-eval-after-load 'comp
        (add-to-list 'native-comp-driver-options "-mmacosx-version-min=11"))
      (with-eval-after-load 'comp-run
        (require 'comp)))
  #+END_SRC

* Package Initialization
** Package Manager

   Ensure the org repository and melpa are searched for packages.

   Circa Jun 2026, I was having issues syncing against the official elpa
   and melpa end points.  Hence MirrorService.

   #+BEGIN_SRC emacs-lisp
     (require 'package)

     ;; Use the UK MirrorService mirror for GNU ELPA; keep the archive
     ;; named "gnu" so package priorities and dependencies still work.
     ;; package.el can refresh this mirror over HTTP; the HTTPS endpoint
     ;; currently fails in Emacs even though curl can fetch it.
     (setq package-archives
           (cons '("gnu" . "http://www.mirrorservice.org/sites/elpa.gnu.org/packages/")
                 (assoc-delete-all "gnu" package-archives)))
     ;; Use MELPA's official MirrorService mirror too; the direct
     ;; melpa.org HTTPS endpoint fails from Emacs on this network.
     (setq package-archives
           (cons '("melpa" . "http://www.mirrorservice.org/sites/melpa.org/packages/")
                 (assoc-delete-all "melpa" package-archives)))
     ;; Drop the built-in NonGNU archive: elpa.nongnu.org has the same
     ;; GnuTLS pull failure here, and the configured non-built-ins are
     ;; available from MELPA/GNU/Org.
     (setq package-archives (assoc-delete-all "nongnu" package-archives))
     (unless (assoc-default "org" package-archives)
       (add-to-list 'package-archives '("org" . "https://orgmode.org/elpa/") t))

     ;; Prefer MELPA when a package appears in multiple archives; keep
     ;; Org and GNU available as lower-priority fallbacks.
     (setq package-archive-priorities
           '(("melpa" . 30)
             ("org" . 20)
             ("gnu" . 5)))

     (package-initialize)
   #+END_SRC

   Use =M-x package-refresh-contents= to reload the list of packages as needed.

** Native-compilation warming

   There is no standard "pre-compile everything I'll use" command, so here's a
   small helper. Run it headlessly during provisioning so a fresh machine's first
   interactive Emacs has little left to JIT-compile (which is otherwise noisy and,
   before packages are compiled, can trip transient errors):

   : emacs --batch -l init.el --eval '(jwm/warm-native-comp)'

   It compiles all installed packages plus the core/bundled lisp this config has
   actually loaded (eglot, editorconfig, …) that =package-user-dir= doesn't cover,
   then BLOCKS until the async compilation finishes. A small residual — modules
   loaded only on specific interactive actions (e.g. checkdoc) — still JIT-compiles
   on first use; a batch init can't walk those paths.

   Only lisp that was loaded compiled (=.elc=) is added: Emacs uses native code
   just for those, so compiling a file it loads as source — =init.el= itself —
   wastes the work and warns about every function defined in a deferred block.

   #+BEGIN_SRC emacs-lisp
     (defun jwm/warm-native-comp ()
       "Native-compile installed packages + the lisp this config has loaded.
     Intended for headless provisioning; blocks until compilation completes."
       (when (native-comp-available-p)
         (let ((files (directory-files-recursively package-user-dir "\\.el\\'")))
           (dolist (entry load-history)
             (let* ((f (car entry))
                    (base (and (stringp f) (string-suffix-p ".elc" f)
                               (file-name-sans-extension f)))
                    (src (and base (seq-find #'file-exists-p
                                             (list (concat base ".el")
                                                   (concat base ".el.gz"))))))
               (when src (push src files))))
           (native-compile-async (delete-dups files))
           ;; native-compile-async doesn't fill its queue instantly — settle first.
           (sleep-for 3)
           ;; Emacs 31 renamed comp-async-runnings to comp--async-runnings.
           (let ((runnings (seq-find #'fboundp '(comp--async-runnings
                                                 comp-async-runnings))))
             (while (or comp-files-queue
                        (and runnings (> (or (funcall runnings) 0) 0)))
               (sleep-for 1))))))
   #+END_SRC

** Terminal and Shell Settings

   Bash's default behavior is sufficient for Emacs subprocess use:
   non-interactive shells (=bash -c=) don't read =.bashrc= or
   =.bash_profile= by default, so there's no startup overhead to
   eliminate. =shell-command-switch= stays at its default =-c=.

   Historical note: this section previously housed an extensive
   zsh-specific optimization that overrode =shell-command-switch=
   to =-cf= (no-rcs mode) because zsh /does/ read =.zshenv= even
   in non-interactive mode. With the migration to bash, that
   workaround is unnecessary; the helper =jwm/shell-is-zsh-p= is
   gone with it.

   Wiegley's =esh-toggle= gives a quick Eshell toggle. With a prefix
   argument it sends Eshell a =cd= command for the current buffer's
   =default-directory=, which is the shell-from-buffer behavior I
   want. His helper lives in his repo as local code, so I keep the
   same behavior here with the usual =jwm/= prefix.

   #+BEGIN_SRC emacs-lisp
     (use-package eshell
       :ensure nil
       :commands eshell
       :bind ("C-x C-z" . jwm/eshell-toggle)
       :preface
       (defvar jwm/eshell-toggle-window-configuration nil
         "Window configuration before toggling to Eshell.")

       (defun jwm/eshell-prompt-path ()
         "Return a compact current directory for the Eshell prompt."
         (let* ((path (directory-file-name (abbreviate-file-name (eshell/pwd))))
                (parts (split-string path "/" t))
                (last-part (car (last parts)))
                (parent-part (cadr (reverse parts)))
                (tail (if (and last-part parent-part (string= last-part "main"))
                          (format "%s/%s" parent-part last-part)
                        last-part)))
           (cond
            ((<= (length path) 32) path)
            ((and parent-part (string= last-part "main")) tail)
            ((string-prefix-p "~/" path)
             tail)
            ((> (length parts) 2)
             (format "/.../%s" tail))
            (t path))))

       (defun jwm/eshell-prompt ()
         "Return a compact Eshell prompt."
         (concat (jwm/eshell-prompt-path)
                 (unless (eshell-exit-success-p)
                   (format " [%d]" eshell-last-command-status))
                 (if (= (file-user-uid) 0) " # " " $ ")))

       (defun jwm/eshell-toggle (&optional make-cd)
         "Toggle Eshell.
     With a prefix argument, cd Eshell to the current buffer's directory."
         (interactive "P")
         (if (eq major-mode 'eshell-mode)
             (jwm/eshell-toggle-return)
           (jwm/eshell-toggle-goto make-cd)))

       (defun jwm/eshell-toggle-return ()
         "Return from Eshell to the saved window configuration."
         (interactive)
         (if (window-configuration-p jwm/eshell-toggle-window-configuration)
             (progn
               (set-window-configuration jwm/eshell-toggle-window-configuration)
               (setq jwm/eshell-toggle-window-configuration nil)
               (bury-buffer "*eshell*"))
           (bury-buffer)))

       (defun jwm/eshell-toggle-goto (make-cd)
         "Switch to Eshell, optionally changing to the current buffer directory."
         (let* ((origin-directory (file-name-as-directory
                                   (expand-file-name
                                    (or default-directory
                                        (bound-and-true-p list-buffers-directory)))))
                (eshell-buffer (get-buffer "*eshell*"))
                (cd-command (and make-cd
                                 (format "cd %s" (shell-quote-argument origin-directory)))))
           (setq jwm/eshell-toggle-window-configuration (current-window-configuration))
           (if eshell-buffer
               (switch-to-buffer-other-window eshell-buffer)
             (let ((default-directory origin-directory))
               (eshell)))
           (when (or cd-command make-cd)
             (goto-char (point-max)))
           (when cd-command
             (insert cd-command)
             (eshell-send-input))))
       :config
       (setq eshell-prompt-function #'jwm/eshell-prompt)

       (defun eshell/ll (&rest args)
         "List files in long form, preserving shell muscle memory."
         (apply #'eshell/ls "-lF" args))

       (defun eshell/la (&rest args)
         "List all files in long form, preserving shell muscle memory."
         (apply #'eshell/ls "-laF" args))

       (defun eshell/lt (&rest args)
         "List all files in long time order, preserving shell muscle memory."
         (apply #'eshell/ls "-laFtr" args))

       (defun eshell/lsd (&rest args)
         "List directories in long form."
         (let ((targets args))
           (unless targets
             (dolist (file (directory-files default-directory nil directory-files-no-dot-files-regexp))
               (when (file-directory-p file)
                 (push file targets)))
             (setq targets (nreverse targets)))
           (when targets
             (apply #'eshell/ls "-ldF" targets))))

       (defun eshell/path (&rest args)
         "Print PATH entries one per line.
      The -l option is accepted for shell muscle memory."
         (dolist (arg args)
           (unless (string= arg "-l")
             (user-error "Unsupported path option: %s" arg)))
         (dolist (entry (split-string (or (getenv "PATH") "") path-separator t))
           (eshell-print (concat entry "\n"))))

       (dolist (command '(eshell/ll eshell/la eshell/lt eshell/lsd))
         (put command 'eshell-no-numeric-conversions t)
         (put command 'eshell-filename-arguments t)))
   #+END_SRC

** Buffer cleanup for removed worktrees

   When a git worktree is removed, Emacs keeps any buffers that were
   visiting files inside it — now pointing at paths that no longer
   exist. =project-kill-buffers= is the wrong tool *after* removal:
   the worktree directory is gone, so project.el walks up to the next
   enclosing =.git= and scopes too broadly.

   Two helpers cover the two moments:

   - =jwm/kill-buffers-under-dir= — run on a worktree root *before*
     =git worktree remove=, to close exactly that worktree's buffers
     (project-independent, so it works even when point isn't in one
     of them).
   - =jwm/kill-buffers-of-missing-files= — the recovery path *after* a
     removal, killing any buffer whose backing file is already gone.

   Both skip modified buffers so unsaved edits are never lost silently,
   and skip remote (TRAMP) files to avoid stat-ing over the network.

   #+BEGIN_SRC emacs-lisp
     (defun jwm/kill-buffers-of-missing-files ()
       "Kill unmodified file buffers whose backing file no longer exists.
     Skips modified buffers and remote (TRAMP) files."
       (interactive)
       (let ((n 0))
         (dolist (buf (buffer-list))
           (let ((file (buffer-file-name buf)))
             (when (and file
                        (not (file-remote-p file))
                        (not (file-exists-p file))
                        (not (buffer-modified-p buf)))
               (kill-buffer buf)
               (setq n (1+ n)))))
         (message "Killed %d buffer(s) visiting missing files" n)))

     (defun jwm/kill-buffers-under-dir (dir)
       "Kill unmodified file buffers whose file lives under DIR.
     Use this on a worktree root *before* `git worktree remove'."
       (interactive "DKill file buffers under directory: ")
       (let ((root (file-name-as-directory (expand-file-name dir)))
             (n 0))
         (dolist (buf (buffer-list))
           (let ((file (buffer-file-name buf)))
             (when (and file
                        (string-prefix-p root (expand-file-name file))
                        (not (buffer-modified-p buf)))
               (kill-buffer buf)
               (setq n (1+ n)))))
         (message "Killed %d buffer(s) under %s" n root)))

   #+END_SRC

** Adjust font size to display resolution

   With the broad range of pixels per inch offered by laptops, 4k displays, etc.
   a mechanism to adjust the font size based on screen resolution is helpful.
   [[https://github.com/WJCFerguson][WJCFerguson]] seems to have provided just such a capability via [[https://github.com/WJCFerguson/textsize][textsize]]

   The customizations help me with my specific font size choices;

   My macbook with retina display at max resolution reports 344 mm wide by 2056 pixels
   for a pixel density of 0.17.  My preference is for the 17 point Hack font,
   which requires an offset of 2.
   My 27inch Dell 4k monitor reports 598 mm wide by 3840 pixel width == pixel density of 0.16.
   I seem to prefer Hack-20 on the 27inch Dell.
   My 32inch Dell 4k monitor reports 701 mm wide by 3840 pixel width for a pixel density of 0.18.
   I prefer Hack-24 there, which requires the offset of 9.
   The ultra-wide monitors at the office report 801 mm wide by pixel width 3440 == density: 0.23
   I prefer Hack-18 there, achieved by switching on pixel pitch.

   #+begin_src emacs-lisp
     (use-package textsize
       :init (textsize-mode)
       :config
       (customize-set-variable 'textsize-monitor-size-thresholds
                               '((0 . 2) (400 . 5) (650 . 6)))
       (customize-set-variable 'textsize-pixel-pitch-thresholds
                               '((0 . 3) (0.22 . -6))))


   #+end_src

   To help understand and manage the choices made by [[https://github.com/WJCFerguson/textsize][textsize]], here is a helper
   function that dumps to the message buffer some relevant info.
   Invoke this from the frame of interest, ie, I have 3 physical monitors in my home setup;
   running this on a frame on each monitor gives the results for that single monitor.

   #+begin_src emacs-lisp
     (defun jwm/dump-frame-textsize-metrics ()
       "Dump selected frame metrics from the currently selected frame to the *Message* buffer.
     Intended to be helpful for debugging the choices textsize makes for a given monitor/display."
       (interactive)
       (let ((f (selected-frame)))
         (message "emacs frame geometry: X Y WIDTH HEIGHT: %s" (frame-monitor-attribute 'geometry f))
         (message "emacs monitor size WIDTH HEIGHT mm: %s" (frame-monitor-attribute 'mm-size f))
         (message "textsize monitor size  mm: %d" (textsize--monitor-size-mm f))
         (message "textsize monitor size pix: %d" (textsize--monitor-size-px f))
         (message "pixel pitch %.02f" (textsize--pixel-pitch f))
         (message "textsize default points %d" textsize-default-points)
         (message "textsize frame offset %d"
                  (or (frame-parameter f 'textsize-manual-adjustment) 0))
         (message "pixel pitch adjustment %d"
                  (textsize--threshold-offset textsize-pixel-pitch-thresholds
                                              (textsize--pixel-pitch f)))
         (message "monitor size adjustment %d"
                  (textsize--threshold-offset textsize-monitor-size-thresholds
                                              (textsize--monitor-size-mm f)))
         (message "text size chosen: %d" (textsize--point-size f))
         (message "default-font: WIDTHxHEIGHT %dx%d" (default-font-width)(default-font-height))
         (message "resultant text area in chars WIDTHxHEIGHT %dx%d"
                  (frame-width f)(frame-height f))
         (message "default face font %s" (face-attribute 'default :font))
         )
       nil)
   #+end_src

   To make a session-permanent adjustment on a per-frame basis, invoke this function
   #+begin_src emacs-lisp
     (defun jwm/adjust-textsize-frame-offset (arg)
       "Supply a per-frame, persistent integer offset (positive or negative) to the textsize font size calculation.
     In effect, adjusts the pixel size of the frame font up or down by the prefix value."
       (interactive "P")
       (let ((f (selected-frame))
             (offset arg))
         (textsize-modify-manual-adjust f offset)))
   #+end_src

** Corfu

   In-buffer completion popup. The minibuffer side of the stack
   (vertico/orderless/consult) had no in-buffer counterpart, so
   =completion-at-point= used to dump candidates into a =*Completions*=
   buffer. Corfu renders them as a popup at point instead.

   =corfu-auto= pops the completion popup on its own as I type. For an
   explicit trigger, =C-M-i= (=completion-at-point=) always works.
   =tab-always-indent= is =complete=, so TAB /can/ complete too — but
   only when the line is already correctly indented. In Python the
   indentation command cycles indent levels and usually consumes TAB
   first, so TAB-to-complete is unreliable there; use =C-M-i=.

   =orderless= is already in =completion-styles=, so its fuzzy matching
   works inside the popup for free. In Python (and other Eglot) buffers
   the language server's candidates flow through the same popup.

   =corfu-popupinfo-mode= shows the highlighted candidate's docstring /
   signature (from Eglot/ty) in a side popup.

   #+BEGIN_SRC emacs-lisp
     (use-package corfu
       :if (package-installed-p 'corfu)
       :custom
       (corfu-auto t)            ; popup appears automatically as you type
       (corfu-auto-delay 0.6)    ; ...after this idle delay
       (corfu-auto-prefix 2)     ; ...once 2 chars are typed
       (corfu-cycle t)           ; wrap around the candidate list
       :config
       (global-corfu-mode)
       (corfu-popupinfo-mode))   ; show docstring beside the candidate
   #+END_SRC

*** Worktrees should preserve context, not create more of it

    A branch in its own checkout is valuable because it lets me keep two
    pieces of work genuinely separate.  The practical failure mode is then
    remembering where I put each checkout.  The convention
    =<repo>.worktrees/<branch-slug>= makes that location predictable from the
    primary repo, so creation should begin there instead of asking me to
    reconstruct a path every time.

    Magit already owns the Git operation.  This reader only gives Magit the
    convention's starting point; the minibuffer is deliberately left editable
    for the exceptional case.  Starting from the primary worktree matters when
    I create a second branch while already visiting an existing worktree.

    #+BEGIN_SRC emacs-lisp
      (defun jwm/worktree-container-directory (primary-worktree)
        "Return the `.worktrees' sibling directory for PRIMARY-WORKTREE."
        (let* ((worktree (directory-file-name
                          (expand-file-name primary-worktree)))
               (parent (file-name-directory worktree))
               (name (file-name-nondirectory worktree)))
          (expand-file-name (concat name ".worktrees/") parent)))

      (defun jwm/magit-read-worktree-directory (prompt branch)
        "Read a new worktree directory using Jeff's sibling convention.
      BRANCH supplies the slash-safe default leaf name."
        (let* ((primary-worktree (caar (magit-list-worktrees)))
               (container (jwm/worktree-container-directory primary-worktree))
               (leaf (and branch (string-replace "/" "-" branch))))
          (make-directory container t)
          (read-directory-name prompt container nil nil leaf)))
    #+END_SRC

*** Git, rather than the filesystem, decides what is a worktree

    A directory under =.worktrees= is not necessarily a usable worktree: it
    may have been removed, moved, or simply left behind.  Git's worktree
    registry is the authority.  Using it keeps the chooser honest and means
    this workflow still works if the storage convention changes later.

    =jwm/magit-switch-worktree= is the fast return path when I know I want to
    resume another branch but do not want to navigate there file by file.  It
    opens Magit status, preserving the same Git-centered starting point as
    =C-x g= then =Z g=.

    #+BEGIN_SRC emacs-lisp
      (defun jwm/git-worktree-paths-from-porcelain (output)
        "Return registered worktree paths from Git porcelain OUTPUT."
        (let (paths)
          (dolist (line (split-string output "\n" t))
            (when (string-prefix-p "worktree " line)
              (push (file-name-as-directory (substring line 9)) paths)))
          (nreverse paths)))

      (defun jwm/git-worktree-paths ()
        "Return the worktrees registered for the current Git repository."
        (with-temp-buffer
          (unless (zerop (process-file "git" nil t nil "worktree" "list" "--porcelain"))
            (user-error "Could not list Git worktrees"))
          (jwm/git-worktree-paths-from-porcelain (buffer-string))))

      (defun jwm/magit-switch-worktree ()
        "Select a Git-registered worktree and show its Magit status buffer."
        (interactive)
        (require 'magit)
        (let ((worktree (completing-read "Visit worktree: "
                                         (jwm/git-worktree-paths) nil t)))
          (magit-status-setup-buffer worktree)))
    #+END_SRC

*** Where to Look for Org Info

    Derived rather than hardcoded, so an Emacs started inside a =jwm-todo=
    worktree resolves the agenda against *that* worktree — the registry is
    tracked at the repo root, so each worktree carries its own.  Searching for
    the registry is also what makes the derivation safe: started anywhere else,
    including another repo entirely, it falls back to the corpus rather than
    adopting an unrelated project root.

    =vc-root-dir= is the obvious way to find the root and does not work here:
    it needs a buffer visiting a file, and during startup that is =*scratch*=,
    so it returns nil no matter where Emacs was launched.

            (let ((root (locate-dominating-file
                         default-directory "org-agenda-files-list")))
              (if root
                  (abbreviate-file-name (directory-file-name root))
                "~/jwm/todo")))

**** Agenda Files

     Recall that if =org-agenda-files= is a single file name, then that symbol names a file which is read
     for the list of agenda files to manage.

     Further, the functions =org-agenda-file-to-front= and =org-remove-file= can be used to manage that list.

     This is the mechanism I want to utilize here.

     The pointer file lives at the root of the corpus repo (=org-directory=), tracked
     in git, rather than under =~/.emacs.d=.  A registry outside the corpus is one file
     of absolute paths shared by every checkout, so an Emacs working in a =jwm-todo=
     worktree resolves the agenda against the *main* checkout — a silent wrong answer.
     Inside the repo, each worktree carries its own registry describing its own files.

     Entries are therefore bare filenames, which resolve against =org-directory=;
     agenda files living outside the repo use =~=-rooted paths.  A session working in a
     worktree binds =org-directory= to that worktree's root — =(or (vc-root-dir)
     default-directory)= — and re-derives =org-agenda-files= from it.

     Being tracked, the file is present in every checkout, so the old first-run
     =make-empty-file= guard against =org-read-agenda-file-list: Opening input file: No
     such file or directory= is gone with it.

     #+BEGIN_SRC emacs-lisp
       (defvar jwm/org-agenda-files-list
         (expand-file-name "org-agenda-files-list" org-directory)
         "File naming the files `org-agenda' scans, one path per line.")

       (setq org-agenda-files jwm/org-agenda-files-list)
     #+END_SRC

  - syntax-highlighted code inside fenced blocks (python, elisp, etc.
    get their native font-lock inside ~```lang~ fences)
  - progressively larger heading fonts
  - =gfm-mode= for =.md= since most real-world markdown is GitHub-flavored

  Reading/editing only; live preview and export are out of scope for this
  pilot.

** gptel tools

   Tools come from [[https://github.com/skissue/llm-tool-collection][llm-tool-collection]], registered by tag as jwiegley
   does: =filesystem=, =editing= and =introspection=.  Its file search
   (=grep=) runs as an asynchronous process, so no tool blocks Emacs
   while it works.  The =system= tools (=bash=, =eval_elisp=) are not
   registered.  The tools that create, edit or replace a file, and
   =replace_buffer=, ask before they run; =edit_buffer= does not.

   Tools are chosen per request: =gptel-use-tools= keeps its default
   and =gptel-tools= stays empty, so no tool is sent until one is

   for claude-code-ide.

   =jwm/required-vc-packages= (see Use-Package) and installed by
   =just install-packages=.

         (apply #'gptel-make-tool tool)))

** live-emacs — the MCP server for this session

   =live-emacs= lets an external process ask *this running* Emacs what it
   can do, and evaluate elisp in it.  The name states the point: a batch
   Emacs is a different Emacs, answering for a config it does not have —
   =emacs -Q= rejects =NEXT= for want of my =org-todo-keywords=, and even
   a full =--batch= init costs 3.38s per call against roughly 10ms here.

   Three tools are read-only introspection.  The fourth, =elisp_eval=,
   evaluates arbitrary forms, so a client that can reach this server can
   do anything Emacs can do, including writing files.  That is the design
   rather than an oversight: Emacs's API *is* elisp, so a caller that can
   evaluate can also discover, which is what makes a hand-maintained
   capability manifest unnecessary — and a manifest that drifts from
   reality is the failure this avoids.

   **It is opt-in and stays that way.**  Loading Emacs defines the tools
   and the commands; it does not start a server.  =M-x jwm/mcp-live-emacs-start=
   is a deliberate act, because the alternative is every Emacs I launch
   exposing an eval endpoint.

   Stdio transport uses the =mcp-server-lib= package helper
   =emacs-mcp-stdio.sh=. The package keeps its source copy inside the
   installed package directory; =M-x mcp-server-lib-install= copies
   that package-provided helper to
   =~/.emacs.d/emacs-mcp-stdio.sh= and marks it executable. That
   global helper is acceptable package state when this package is part
   of the config, but its provenance and lifecycle should be explicit:
   run =just mcp-stdio-helper-status= to compare the global target to
   the package source, use =M-x mcp-server-lib-uninstall= to remove it,
   and repeat the install step on other machines if this MCP stdio
   path becomes part of the durable setup.

   #+BEGIN_SRC emacs-lisp
     (use-package mcp-server-lib
       :commands (mcp-server-lib-install
                  mcp-server-lib-uninstall
                  mcp-server-lib-start
                  mcp-server-lib-stop
                  mcp-server-lib-register-tool)
       :init
       (defvar jwm/mcp-live-emacs-server-id "live-emacs"
         "MCP server identifier for this Emacs session's tools.")

       (defvar jwm/mcp-live-emacs-tools-registered nil
         "List of `live-emacs' MCP tool IDs registered in this session.")

       (defun jwm/mcp-live-emacs--json-false (value)
         "Return VALUE as a JSON boolean value."
         (if value t :json-false))

       (defun jwm/mcp-live-emacs--first-doc-line (symbol)
         "Return the first documentation line for SYMBOL, or nil."
         (when (symbolp symbol)
           (let ((doc (documentation symbol t)))
             (when (stringp doc)
               (car (split-string doc "\n" t))))))

       (defun jwm/mcp-live-emacs--register-tool (handler &rest properties)
         "Register HANDLER with PROPERTIES once for this Emacs session."
         (require 'mcp-server-lib)
         (when (eq jwm/mcp-live-emacs-tools-registered t)
           (setq jwm/mcp-live-emacs-tools-registered '("emacs_state_summary")))
         (let ((id (plist-get properties :id)))
           (unless (member id jwm/mcp-live-emacs-tools-registered)
             (apply #'mcp-server-lib-register-tool handler properties)
             (add-to-list 'jwm/mcp-live-emacs-tools-registered id))))

       (defun jwm/mcp-live-emacs-state-summary ()
         "Return read-only JSON metadata about the current Emacs session."
         (require 'json)
         (let* ((window (selected-window))
                (buffer (and (window-live-p window)
                             (window-buffer window)))
                (project-root
                 (when (and buffer
                            (fboundp 'project-current)
                            (fboundp 'project-root))
                   (with-current-buffer buffer
                     (let ((project (project-current nil)))
                       (when project
                         (expand-file-name (project-root project))))))))
           (with-current-buffer (or buffer (current-buffer))
             (json-encode
              `(("buffer" . ,(buffer-name))
                ("major_mode" . ,(symbol-name major-mode))
                ("file" . ,buffer-file-name)
                ("default_directory" . ,default-directory)
                ("project_root" . ,project-root)
                ("emacs_version" . ,emacs-version)
                ("read_only" . t))))))

       (defun jwm/mcp-live-emacs-key-binding (key)
         "Return read-only JSON metadata about KEY's active binding.

     MCP Parameters:
       key - Emacs key sequence accepted by `kbd', for example C-x c g"
         (require 'json)
         (condition-case err
             (let* ((key-vector (kbd key))
                    (command (key-binding key-vector t))
                    (command-symbol (and (symbolp command) command))
                    (command-name
                     (cond
                      ((symbolp command) (symbol-name command))
                      ((null command) nil)
                      (t (prin1-to-string command))))
                    (command-type
                     (cond
                      ((symbolp command) "symbol")
                      ((null command) "unbound")
                      (t "function"))))
               (json-encode
                `(("key" . ,key)
                  ("normalized_key" . ,(key-description key-vector))
                  ("command" . ,command-name)
                  ("command_type" . ,command-type)
                  ("command_doc_summary" . ,(jwm/mcp-live-emacs--first-doc-line command-symbol))
                  ("read_only" . t))))
           (error
            (json-encode
             `(("key" . ,key)
               ("error" . ,(error-message-string err))
               ("read_only" . t))))))

         "Return read-only JSON metadata about LIBRARY or feature name.

                (loaded (and feature (featurep feature)))

              ("feature_loaded" . ,(jwm/mcp-live-emacs--json-false loaded))
              ("library_path" . ,path)
              ("read_only" . t)))))

         "Evaluate FORM and return its value's printed representation.

     A failing form is reported with `mcp-server-lib-tool-throw', which the

     arrives instead as a JSON-RPC error, so the two stay distinguishable
     at the protocol layer without wrapping the value in an envelope.

           (error

** claude-code-ide

   Coding agents run as the =claude= CLI inside Emacs, through
   [[https://github.com/manzaltu/claude-code-ide.el][claude-code-ide]]: each session is a =ghostel= buffer tied to its
   =project.el= project, and a project can hold several.  My Claude Code
   settings, skills and =CLAUDE.md= files apply unchanged.  Through the
   IDE WebSocket connection Claude opens files here, reads the selection
   and diagnostics, and proposes edits as ediff sessions.  Decided in
   workspace spec 017.

   It loads wherever =claude= is on =PATH=, not just on the personal
   machine.  Claude Code's permission prompts stay on: nothing adds
   =--dangerously-skip-permissions= to =claude-code-ide-cli-extra-flags=.
   =claude-code-ide-mcp-allowed-tools= keeps its default, =auto=, which
   pre-approves only the read-only tools on its =emacs-tools= server
   (xref, imenu, project info, tree-sitter); everything else, including
   the IDE connection's =executeCode=, goes through Claude Code's normal
   rules.  =claude-code-ide-enable-execute-code= keeps its default, on,
   although the WebSocket accepts any local client until
   manzaltu/claude-code-ide.el#225 merges.

   Its servers listen on random free ports, and =live-emacs= above speaks
   stdio through =emacsclient=, so the two never compete for a port.

   claude-code-ide is not on MELPA.  Its README installs it with
   use-package =:vc=, but =:vc= clones whenever Emacs loads without the
   package, and loading this config never installs anything.  So it is
   listed in =jwm/required-vc-packages= (see Use-Package) and installed by
   =just install-packages= (or =M-x package-vc-install=), which also pulls
   =websocket=, =web-server= and =transient= from the archives.

   Sessions run in [[https://github.com/dakra/ghostel][ghostel]], a terminal built on libghostty (the engine
   behind Ghostty), which claude-code-ide recommends as its backend.
   ghostel needs a native module.  The first time it loads it asks to
   download a prebuilt one (=ghostel-module-auto-install= keeps its
   default, =ask=), so no machine builds one.  ghostel replaced vterm,
   which also served =M-x vterm=; =M-x ghostel= does that now.

   - Path: =~/.emacs.d/elpa/ghostel-*/=, the module saved inside it.
   - Provenance: MELPA for the package; the module is a prebuilt binary
     from the =dakra/ghostel= GitHub releases, checked by release
     version, not by checksum.
   - Teardown: =M-x package-delete RET ghostel=, which removes the module
     with the package directory.
   - Replication: each machine runs =just install-packages= and accepts
     the download prompt once.

   #+BEGIN_SRC emacs-lisp
     (use-package ghostel
       :commands (ghostel ghostel-project))

     (use-package claude-code-ide
       :if (executable-find "claude")
       :bind (:map jwm/ai-map ("c" . claude-code-ide-menu))
       :custom
       (claude-code-ide-terminal-backend 'ghostel)
       (claude-code-ide-cli-extra-flags
        (concat "--settings " (shell-quote-argument jwm/claude-code-hooks-file)))
       :config
       (claude-code-ide-emacs-tools-setup)
       (advice-add 'claude-code-ide--create-terminal-session
                   :around #'jwm/claude-code-ide-add-buffer-name))
   #+END_SRC

*** Session status

    claude-code-ide reports nothing about what a session is doing, so
    Claude Code's own hooks tell Emacs: a session that needs permission
    or is waiting for input, finishes a turn, starts working, or ends
    calls =emacsclient=, and Emacs shows it.  This follows
    [[https://github.com/stevemolitor/claude-code.el][claude-code.el]], which is not installed; its receiver is copied below.
    Decided in workspace spec 017.

    The hooks only reach sessions claude-code-ide starts: they are in
    =claude-code-ide-hooks.json=, tangled beside =init.el= and passed with
    =--settings=, so global Claude settings are untouched and a =claude=
    started outside Emacs sends nothing.  Each hook command needs
    =CLAUDE_BUFFER_NAME=, which advice on claude-code-ide's private
    =claude-code-ide--create-terminal-session= puts in the session's
    environment.  If a package update changes that function, sessions
    still start; the variable is missing and the hooks do nothing.

    Hook commands send their output to =/dev/null=: =emacsclient= prints
    the result, and a =UserPromptSubmit= hook's output is added to
    Claude's context.  They always exit 0, so an Emacs without a server
    (only the windowed one runs =server-start=) never blocks a session.

    #+BEGIN_SRC json :tangle claude-code-ide-hooks.json
      {
        "hooks": {
          "Notification": [
            {
              "matcher": "permission_prompt",
              "hooks": [{"type": "command", "command": "[ -z \"$CLAUDE_BUFFER_NAME\" ] || emacsclient --eval \"(jwm/claude-code-handle-hook 'permission \\\"$CLAUDE_BUFFER_NAME\\\")\" \"$(cat)\" >/dev/null 2>&1 || true"}]
            },
            {
              "matcher": "idle_prompt",
              "hooks": [{"type": "command", "command": "[ -z \"$CLAUDE_BUFFER_NAME\" ] || emacsclient --eval \"(jwm/claude-code-handle-hook 'idle \\\"$CLAUDE_BUFFER_NAME\\\")\" \"$(cat)\" >/dev/null 2>&1 || true"}]
            }
          ],
          "Stop": [
            {"hooks": [{"type": "command", "command": "[ -z \"$CLAUDE_BUFFER_NAME\" ] || emacsclient --eval \"(jwm/claude-code-handle-hook 'stop \\\"$CLAUDE_BUFFER_NAME\\\")\" \"$(cat)\" >/dev/null 2>&1 || true"}]}
          ],
          "UserPromptSubmit": [
            {"hooks": [{"type": "command", "command": "[ -z \"$CLAUDE_BUFFER_NAME\" ] || emacsclient --eval \"(jwm/claude-code-handle-hook 'prompt \\\"$CLAUDE_BUFFER_NAME\\\")\" \"$(cat)\" >/dev/null 2>&1 || true"}]}
          ],
          "SessionEnd": [
            {"hooks": [{"type": "command", "command": "[ -z \"$CLAUDE_BUFFER_NAME\" ] || emacsclient --eval \"(jwm/claude-code-handle-hook 'session-end \\\"$CLAUDE_BUFFER_NAME\\\")\" \"$(cat)\" >/dev/null 2>&1 || true"}]}
          ]
        }
      }
    #+END_SRC

    The receiver and mode-line pulse are adapted from claude-code.el
    =03199df= (=claude-code-event-hook=, =claude-code-handle-hook=,
    =claude-code--pulse-modeline=), Copyright Stephen Molitor, Apache
    License 2.0.  Its reply path, which returns JSON to Claude Code, is
    left out: these hooks only report status.  The mode line does not
    pulse when a session starts working, since that follows my own
    prompt; a macOS notification is added for "needs permission" and
    "waiting for input".

    #+BEGIN_SRC emacs-lisp
      (defconst jwm/claude-code-hooks-file
        (expand-file-name "claude-code-ide-hooks.json"
                          (file-name-directory
                           (file-truename (or load-file-name buffer-file-name))))
        "Claude Code settings file holding the session status hooks.")

      (defun jwm/claude-code-ide-add-buffer-name (orig-fn buffer-name &rest args)
        "Call ORIG-FN with CLAUDE_BUFFER_NAME=BUFFER-NAME in the environment.
      ARGS are passed through.  Advises `claude-code-ide--create-terminal-session'."
        (let ((process-environment
               (if (stringp buffer-name)
                   (cons (concat "CLAUDE_BUFFER_NAME=" buffer-name)
                         process-environment)
                 process-environment)))
          (apply orig-fn buffer-name args)))

      ;; Adapted from claude-code.el 03199df, Copyright Stephen Molitor,
      ;; Apache License 2.0.
      (defvar jwm/claude-code-event-hook nil
        "Hook run when a Claude Code session hook calls Emacs.
      Each function gets one plist with :type, :buffer-name, :json-data
      and :args.")

      (defvar server-eval-args-left)

      (defun jwm/claude-code-handle-hook (type buffer-name &rest args)
        "Run `jwm/claude-code-event-hook' for event TYPE from BUFFER-NAME.
      Called through `emacsclient --eval' with the hook's JSON as the next
      argument; ARGS and any further arguments are passed along."
        ;; Consume every argument, or the server evaluates the rest as Lisp.
        (let ((json-data (pop server-eval-args-left))
              (extra-args (prog1 server-eval-args-left
                            (setq server-eval-args-left nil))))
          (run-hook-with-args 'jwm/claude-code-event-hook
                              (list :type type
                                    :buffer-name buffer-name
                                    :json-data json-data
                                    :args (append args extra-args)))
          nil))

      (defun jwm/claude-code--pulse-mode-line ()
        "Flash the mode line twice."
        (dolist (delay '(0 0.1 0.2 0.3))
          (run-at-time delay nil #'invert-face 'mode-line)))

      (defun jwm/claude-code--macos-notify (title body)
        "Post a macOS notification with TITLE and BODY."
        (when (and (eq system-type 'darwin) (executable-find "osascript"))
          (make-process
           :name "claude-code-notify"
           :noquery t
           :command (list "osascript"
                          "-e" "on run argv"
                          "-e" "display notification (item 2 of argv) with title (item 1 of argv)"
                          "-e" "end run"
                          title body))))

      (defun jwm/claude-code-session-status (event)
        "Show the session status EVENT from `jwm/claude-code-event-hook'."
        (let* ((type (plist-get event :type))
               (buffer-name (plist-get event :buffer-name))
               (detail (ignore-errors
                         (gethash "message"
                                  (json-parse-string (plist-get event :json-data)))))
               (status (pcase type
                         ('permission "needs permission")
                         ('idle "is waiting for input")
                         ('stop "finished its turn")
                         ('prompt "is working")
                         ('session-end "ended")
                         (_ (format "sent %s" type)))))
          (message "Claude %s: %s%s" status buffer-name
                   (if detail (concat " — " detail) ""))
          (unless (eq type 'prompt)
            (jwm/claude-code--pulse-mode-line))
          (when (memq type '(permission idle))
            (jwm/claude-code--macos-notify
             (concat "Claude " status)
             (if detail (concat buffer-name ": " detail) buffer-name)))))

      (add-hook 'jwm/claude-code-event-hook #'jwm/claude-code-session-status)
    #+END_SRC

** Rust

   Emacs 30's built-in =rust-ts-mode= is the Rust editing mode for this
   configuration.  It becomes active for =.rs= files only after the Rust
   tree-sitter grammar is installed (see the Tree-sitter bootstrap below).

   Eglot already maps both =rust-mode= and =rust-ts-mode= to
   =rust-analyzer=, so this needs no server-program override.  The initial
   formatting interface is explicit rather than format-on-save: =C-c r f=
   asks the language server to format the current buffer and therefore honors
   the Cargo project's Rustfmt policy.

   Emacs's =project-compile= is build-tool-agnostic and otherwise defaults its
   prompt to the global =make -k=.  A buffer-local =compile-command= set in Rust
   buffers pre-fills =cargo check= instead, so =C-x p c= from a =.rs= buffer
   starts from a Cargo command (still editable for =--workspace=, =cargo test=,
   etc.).  Invoked from Dired or a non-Rust buffer it still shows =make -k=.

** Tree-sitter

   Emacs 30 ships =treesit.el= and =-ts-mode= variants for most major
   languages. Grammars must be installed once per machine via
   =M-x treesit-install-language-grammar=. Once a grammar is available,
   =major-mode-remap-alist= redirects the legacy mode to its
   tree-sitter cousin without touching =auto-mode-alist=.

   #+BEGIN_SRC emacs-lisp
     (setq treesit-font-lock-level 4)  ; maximum fontification

     ;; When a grammar is present, remap the legacy mode (for
     ;; languages where a legacy mode exists) or wire auto-mode-alist
     ;; directly (for languages with no legacy mode) to the
     ;; tree-sitter variant.
     (when (treesit-ready-p 'python t)
       (add-to-list 'major-mode-remap-alist '(python-mode . python-ts-mode)))

     (when (treesit-ready-p 'json t)
       (add-to-list 'major-mode-remap-alist '(json-mode . json-ts-mode)))

     (when (treesit-ready-p 'yaml t)
       (add-to-list 'major-mode-remap-alist '(yaml-mode . yaml-ts-mode)))

     ;; toml has no legacy mode in this config; wire the extension
     ;; directly to the tree-sitter mode.

** Python environment (pet)

   [[https://github.com/wyuenho/emacs-pet][PET]] (Python Executable Tracker) discovers the project's
   virtualenv and rebinds the executables that Eglot and the inferior
   Python shell use, per buffer. With uv that means =ty= and =python3=
   resolve to the project's =.venv= instead of whatever happens to be
   on =PATH=.

   =pet-mode= is hooked on =python-base-mode-hook= (parent of both
   =python-mode= and =python-ts-mode=) at depth =-10= so it runs
   /before/ =eglot-ensure= on the same buffer; the executables must be
   set buffer-local before Eglot starts. pet advises Eglot's command
   resolution, so the =("ty" "server")= entry in the Eglot section
   picks up the venv's =ty= automatically.

   Ruff is intentionally left out of pet's reach: a single global
   =ruff= (e.g. =uv tool install ruff=) handles both =ruff-format= and
   =flymake-ruff= for every project. Formatting and lint style stay
   consistent across projects rather than tracking each venv, so
   neither the =ruff-format= reformatter nor =flymake-ruff-program=
   are rerouted.

   This is complementary to =envrc= (direnv) below: envrc applies a
   project's =.envrc= exports, while pet discovers the venv for
   projects that don't have one.

   pet works with a pure-Elisp parser out of the box, and runs faster
   when =dasel= and =sqlite3= are on =PATH=. Use =M-x pet-verify-setup=
   in a Python buffer to confirm what it resolved.

   #+BEGIN_SRC emacs-lisp
     (use-package pet
       :config
       (add-hook 'python-base-mode-hook #'pet-mode -10))
   #+END_SRC

*** PostgreSQL tools for Claude sessions

    Claude sessions started by claude-code-ide (see =** claude-code-ide=)
    get these tools on its =emacs-tools= server: =pg_describe_schema=,
    =pg_run_query=, =pg_explain= and =pg_run_write=.  They reach whatever
    database the session project's mise environment points at; host,
    port, database and =PGPASSFILE= come from that environment.  The read
    tools run as the login named by its =PGUSER_AGENT=, =pg_run_write= as
    the login named by its =PGUSER_AGENT_WRITE=.  With the variable unset a
    tool refuses and names it; there is no fallback to =PGUSER= or to the
    other login.  gptel gets none of them.  Decided in workspace spec 017
    (emacs-config #40, #41, #43).

    The login alone decides what Claude may do: these tools add no
    read-only guard and no confirmation.  A write commits like any other
    statement; =pg_run_write= sends all its statements in one transaction
    (=psql --single-transaction=), so an error leaves none of them
    applied.  There is no DDL tool: schema changes stay Sqitch changes, and
    the writer login has no rights to make them.  Because
    =claude-code-ide-mcp-allowed-tools= is =auto=, each tool is pre-approved
    in every session claude-code-ide starts.

    Each call is its own =psql -X --csv= process, never my =*SQL*= session.
    The SQL goes in on stdin, unchanged.  Limits: =statement_timeout= 30 s
    and =lock_timeout= 5 s through =PGOPTIONS=; a 35 s Emacs timer kills
    =psql=; Claude sees at most 200 rows and 64 KB, with a line saying the
    result was cut off.

    claude-code-ide's tool server expects a tool to return its answer, so
    a database call would freeze Emacs until =psql= exits.  Instead the
    tool starts =psql=, leaves the HTTP request open with web-server's
    =:keep-alive=, and answers it from the callback.  The request object
    comes from advice on the private
    =claude-code-ide-mcp-http-server--handle-post=; if an update renames
    it, the tools still work but Emacs waits for each answer.

  #+begin_src emacs-lisp
    (defconst jwm/pg-tool-max-rows 200
      "CSV rows, after the header, a PostgreSQL tool shows Claude.")

    (defconst jwm/pg-tool-max-bytes (* 64 1024)
      "Bytes of psql output a PostgreSQL tool shows Claude.")

    (defconst jwm/pg-tool-timeout 35
      "Seconds before Emacs kills a PostgreSQL tool's psql.")

    (defconst jwm/pg-tool-pgoptions "-c statement_timeout=30s -c lock_timeout=5s"
      "Server-side limits for every PostgreSQL tool call.")

    (defun jwm/pg-tool-environment (dir login-var)
      "Return (PSQL . ENVIRONMENT) for running psql as one of DIR's agent logins.
    LOGIN-VAR names the mise variable holding the login.  Signal
    `user-error' when DIR's mise environment does not set it."
      (let* ((default-directory dir)
             (env (with-temp-buffer
                    (when (and (executable-find "mise")
                               (eq 0 (call-process "mise" nil '(t nil) nil
                                                   "env" "--json")))
                      (json-parse-string (buffer-string) :object-type 'alist))))
             (user (alist-get (intern login-var) env))
             (psql (let ((exec-path (append (split-string (or (alist-get 'PATH env) "")
                                                          path-separator t)
                                            exec-path)))
                     (executable-find "psql"))))
        (when (or (null user) (string-empty-p user))
          (user-error "%s is not set in the mise environment of %s; \
    it names the login this PostgreSQL tool runs as"
                      login-var (abbreviate-file-name dir)))
        (unless psql
          (user-error "No psql on %s's PATH" (abbreviate-file-name dir)))
        (cons psql
              (append (list (concat "PGUSER=" user)
                            (concat "PGOPTIONS=" jwm/pg-tool-pgoptions))
                      (mapcar (lambda (kv) (format "%s=%s" (car kv) (cdr kv))) env)
                      process-environment))))

    (defun jwm/pg-tool-cut (output)
      "Return (TEXT . CUT) for OUTPUT, a unibyte string of psql CSV output.
    TEXT keeps the header and `jwm/pg-tool-max-rows' rows, within
    `jwm/pg-tool-max-bytes'; CUT is non-nil when anything was dropped."
      (let ((end (length output))
            (records 0)
            (quoted nil)
            (i 0))
        ;; Count records, not lines: a quoted CSV field may hold newlines.
        (while (< i end)
          (pcase (aref output i)
            (?\" (setq quoted (not quoted)))
            (?\n (unless quoted
                   (setq records (1+ records))
                   (when (> records jwm/pg-tool-max-rows)
                     (setq end (1+ i))))))
          (setq i (1+ i)))
        (setq end (min end jwm/pg-tool-max-bytes))
        (cons (decode-coding-string (substring output 0 end) 'utf-8)
              (< end (length output)))))

    (defun jwm/pg-tool-run (dir login-var sql args callback)
      "Run SQL through psql as the DIR login in LOGIN-VAR, with extra psql ARGS.
    CALLBACK runs exactly once, with the text to show Claude."
      (pcase-let* ((`(,psql . ,env) (jwm/pg-tool-environment dir login-var))
                   (default-directory dir)
                   (process-environment env)
                   (chunks nil)
                   (size 0)
                   (done nil)
                   (timer nil)
                   (proc nil))
        (let ((finish
               (lambda (text)
                 (unless done
                   (setq done t)
                   (when timer (cancel-timer timer))
                   (when (process-live-p proc) (delete-process proc))
                   (funcall callback text))))
              (result
               (lambda (status)
                 (pcase-let ((`(,text . ,cut)
                              (jwm/pg-tool-cut (apply #'concat (reverse chunks)))))
                   (concat (unless (eq status 0)
                             (format "psql exited with status %s:\n" status))
                           text
                           (when cut
                             (format "\n[Result cut off: at most %d rows and %d KB are shown.]"
                                     jwm/pg-tool-max-rows
                                     (/ jwm/pg-tool-max-bytes 1024))))))))
          (setq proc
                (make-process
                 :name "jwm-pg-tool"
                 :command (append (list psql "-X" "--csv" "--no-password"
                                        "-v" "ON_ERROR_STOP=1")
                                  args)
                 :connection-type 'pipe
                 :coding 'no-conversion
                 :noquery t
                 :filter (lambda (_p string)
                           (push string chunks)
                           (setq size (+ size (length string)))
                           ;; Enough to fill the cap: stop reading.
                           (when (> size jwm/pg-tool-max-bytes)
                             (funcall finish (funcall result 0))))
                 :sentinel (lambda (p _event)
                             (when (memq (process-status p) '(exit signal))
                               (funcall finish (funcall result (process-exit-status p)))))))
          (setq timer
                (run-at-time jwm/pg-tool-timeout nil
                             (lambda ()
                               (funcall finish
                                        (format "Error: psql was stopped after %d seconds; \
    the statement did not finish." jwm/pg-tool-timeout)))))
          (process-send-string proc (encode-coding-string sql 'utf-8))
          (process-send-string proc "\n")
          (process-send-eof proc)
          proc)))

    (defvar jwm/pg-tool--request nil
      "The web-server request of the MCP call being handled, or nil.")

    (defun jwm/pg-tool-capture-request (orig request)
      "Call ORIG with REQUEST, keeping REQUEST where the PostgreSQL tools see it."
      (let ((jwm/pg-tool--request request))
        (funcall orig request)))

    (declare-function ws-process "web-server")
    (declare-function ws-body "web-server")
    (declare-function ws-send "web-server")
    (declare-function ws-response-header "web-server")
    (declare-function claude-code-ide-mcp-server-get-session-context "claude-code-ide-mcp-server")
    (declare-function claude-code-ide-make-tool "claude-code-ide-mcp-server")

    (defun jwm/pg-tool-answer (request text)
      "Answer the MCP tools/call REQUEST with TEXT, then close its connection."
      (let ((proc (ws-process request)))
        (when (process-live-p proc)
          (let ((id (alist-get 'id (json-parse-string (ws-body request)
                                                      :object-type 'alist))))
            (ws-response-header proc 200 '("Content-Type" . "application/json"))
            (ws-send proc (json-encode
                           `((jsonrpc . "2.0")
                             (id . ,id)
                             (result (content . [((type . "text") (text . ,text))])))))
            (with-slots (requests) (plist-get (process-plist proc) :server)
              (setq requests (delq request requests)))
            (delete-process proc)))))

    (defun jwm/pg-tool-call-as (login-var sql &rest args)
      "Run SQL as the login in LOGIN-VAR, with extra psql ARGS, for the calling session."
      (let ((dir (or (plist-get (claude-code-ide-mcp-server-get-session-context)
                                :project-dir)
                     default-directory))
            (request jwm/pg-tool--request))
        (if request
            (progn
              (jwm/pg-tool-run dir login-var sql args
                               (lambda (text) (jwm/pg-tool-answer request text)))
              ;; Leave the HTTP request open; the callback answers it.
              (throw 'close-connection :keep-alive))
          (let ((answer nil))
            (jwm/pg-tool-run dir login-var sql args (lambda (text) (setq answer text)))
            (while (not answer)
              (accept-process-output nil 0.1))
            answer))))

    (defun jwm/pg-tool-call (sql &rest args)
      "Run SQL, with extra psql ARGS, as the read login of the calling session."
      (apply #'jwm/pg-tool-call-as "PGUSER_AGENT" sql args))

    (defconst jwm/pg-tool-list-relations-sql
      "select n.nspname as schema, c.relname as name,
           case c.relkind when 'r' then 'table' when 'p' then 'partitioned table'
                          when 'v' then 'view' when 'm' then 'materialized view'
                          when 'f' then 'foreign table' end as kind,
           pg_catalog.obj_description(c.oid, 'pg_class') as comment
    from pg_catalog.pg_class c
    join pg_catalog.pg_namespace n on n.oid = c.relnamespace
    where c.relkind in ('r', 'p', 'v', 'm', 'f')
      and n.nspname not in ('pg_catalog', 'information_schema')
      and n.nspname not like 'pg_toast%'
    order by 1, 2"
      "Every table and view outside the system schemas.")

    (defconst jwm/pg-tool-describe-relation-sql
      "select kind, name, definition from (
      select 1 as part, a.attnum as pos, 'column' as kind, a.attname::text as name,
             pg_catalog.format_type(a.atttypid, a.atttypmod)
             || case when a.attnotnull then ' not null' else '' end
             || coalesce(' default ' || pg_catalog.pg_get_expr(d.adbin, d.adrelid), '')
             as definition
      from pg_catalog.pg_attribute a
      left join pg_catalog.pg_attrdef d on d.adrelid = a.attrelid and d.adnum = a.attnum
      where a.attrelid = :'rel'::regclass and a.attnum > 0 and not a.attisdropped
      union all
      select 2, 0, 'constraint', conname::text, pg_catalog.pg_get_constraintdef(oid)
      from pg_catalog.pg_constraint where conrelid = :'rel'::regclass
      union all
      select 3, 0, 'index', c.relname::text, pg_catalog.pg_get_indexdef(i.indexrelid)
      from pg_catalog.pg_index i join pg_catalog.pg_class c on c.oid = i.indexrelid
      where i.indrelid = :'rel'::regclass
    ) parts order by part, pos, name"
      "Columns, constraints and indexes of the relation in psql variable rel.")

    (defun jwm/pg-tool-describe-schema (&optional table)

        (jwm/pg-tool-call jwm/pg-tool-list-relations-sql)))

    (defun jwm/pg-tools-register ()

       :function #'jwm/pg-tool-describe-schema

** mise support

   mise is my preferred tool for project environments; envrc remains for
   projects that only have a =.envrc=. [[https://github.com/eki3z/mise.el][mise.el]] does for =mise.toml= what
   envrc does for =.envrc=: each buffer gets its project's environment and
   =exec-path=, and org-babel blocks inherit them. Chosen in workspace spec
   015.

   - =mise-trust= is nil: the default asks once, then runs
     =mise trust --all= on every parent directory.
   - The package sets mise's =experimental= setting itself when it is
     off, which would write the dotfiles-managed global config; the
     dotfiles repo sets it instead.
   - envrc and mise each clear the other's buffer-local environment, so
     each buffer gets exactly one. Where a project has both a =.envrc=
     and its own mise config, envrc stands aside and a warning names the
     project. Elsewhere a buffer under a =.envrc= is left to envrc.

   #+begin_src emacs-lisp
     (use-package mise
       :if (executable-find "mise")
       :custom
       (mise-trust nil)
       :config
       (setq mise-exclude-predicate
             (lambda () (or (eq (bound-and-true-p envrc--status) 'on)
                            (mise-default-exclude))))
       :hook (after-init . global-mise-mode))

     (defun jwm/mise-project-config ()
       "Return the closest mise config for `default-directory' that is not global."
       (let ((global (file-name-as-directory (expand-file-name "~/.config/mise"))))
         (with-temp-buffer
           (when (eq 0 (ignore-errors
                         (call-process "mise" nil '(t nil) nil "config" "ls" "--json")))
             (seq-find (lambda (f) (not (string-prefix-p global f)))
                       (mapcar (lambda (c) (expand-file-name (alist-get 'path c)))
                               (ignore-errors
                                 (json-parse-string (buffer-string)
                                                    :object-type 'alist
                                                    :array-type 'list))))))))

     (defvar jwm/mise-envrc-warned nil
       "Projects already warned about having both a .envrc and a mise config.")

     (defun jwm/envrc-yield-to-mise (orig &optional arg)
       "Keep `envrc-mode' off where a project mise config applies too."
       (let ((envrc-dir (and (eql arg 1)
                             (executable-find "mise")
                             (not (file-remote-p default-directory))
                             (locate-dominating-file default-directory ".envrc")))
             (mise-config nil))
         (if (and envrc-dir (setq mise-config (jwm/mise-project-config)))
             (unless (member envrc-dir jwm/mise-envrc-warned)
               (push envrc-dir jwm/mise-envrc-warned)
               (display-warning
                'mise
                (format "%s has both %s.envrc and %s; using mise"
                        (abbreviate-file-name envrc-dir)
                        (abbreviate-file-name envrc-dir)
                        (abbreviate-file-name mise-config))))
           (funcall orig arg))))

     (advice-add 'envrc-mode :around #'jwm/envrc-yield-to-mise)

     (defun jwm/mise-refresh ()
       "Re-read mise for every buffer, then restart the Eglot servers they use.
     Run after changing a project's mise settings, e.g. `just point-tools'."
       (interactive)
       (mise-update-dir t)
       (let ((managed (seq-filter (lambda (b) (with-current-buffer b (eglot-managed-p)))
                                  (buffer-list))))
         (dolist (server (delete-dups (mapcar (lambda (b)
                                                (with-current-buffer b (eglot-current-server)))
                                              managed)))
           (eglot-shutdown server))
         ;; One postgres-language-server daemon serves every project and keeps
         ;; the environment it started with.
         (when (executable-find "postgres-language-server")
           (call-process "postgres-language-server" nil nil nil "stop"))
         (dolist (b managed)
           (with-current-buffer b (eglot-ensure)))))
   #+end_src
